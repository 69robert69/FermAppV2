FROM node:20-alpine

WORKDIR /app

# Copiază fișierele de dependențe și instalează pachetele
COPY package*.json ./
RUN npm install

# Copiază restul codului aplicației
COPY . .

# Expune porturile pentru Expo Metro Bundler
EXPOSE 8081 19000 19001 19002

# Pornește serverul Expo accesibil din afara containerului
CMD ["npx", "expo", "start", "--host", "lan"]
